## 修复方案：解决键盘松开后崩溃问题

### 1. 修复 Player::init - 初始化时资源加载失败保护
**文件**: [Player.cpp](file:///c:/Users/emma/Documents/nyj_lxh_zlf_game/Classes/Core/Player.cpp)

修改 `init()` 函数，添加资源加载失败的处理：
```cpp
bool Player::init()
{
    if (!Sprite::init())
    {
        return false;
    }

    moveSpeed = 150.0f;
    animationFrameRate = 10.0f;
    currentDirection = Direction::DOWN;
    moveDirection = Vec2::ZERO;
    isMoving = false;
    isAnimationPlaying = false;
    currentAnimation = nullptr;
    currentAnimationAction = nullptr;
    lastPressedKey = Direction::NONE;

    // 初始化动画
    initAnimations();

    // 检查资源是否加载成功
    if (!hasValidResources()) {
        CCLOG("WARNING: Player animation resources failed to load!");
        // 创建临时占位符，防止崩溃
        createPlaceholderSprite();
    }

    showStandSprite(Direction::DOWN);

    return true;
}

bool Player::hasValidResources()
{
    // 检查是否有任何方向的站立精灵
    for (int i = 0; i < 4; i++) {
        if (standSprites[i] != nullptr) {
            return true;
        }
    }
    return false;
}

void Player::createPlaceholderSprite()
{
    // 创建占位符纹理
    auto texture = Director::getInstance()->getTextureCache()->addImage("background.png");
    if (texture) {
        setTexture(texture);
        setTextureRect(Rect(0, 0, 70, 120));
    } else {
        // 纯色占位符
        auto placeholder = Sprite::create();
        placeholder->setTextureRect(Rect(0, 0, 70, 120));
        placeholder->setColor(Color3B(128, 128, 128));
        setSpriteFrame(placeholder->getSpriteFrame());
    }
    
    // 确保所有站立精灵有有效值
    for (int i = 0; i < 4; i++) {
        if (standSprites[i] == nullptr) {
            standSprites[i] = getSpriteFrame();
        }
    }
}
```

### 2. 修复 showStandSprite - 添加空指针保护
**文件**: [Player.cpp](file:///c:/Users/emma/Documents/nyj_lxh_zlf_game/Classes/Core/Player.cpp)

```cpp
void Player::showStandSprite(Direction dir)
{
    stopAnimation();
    
    int dirIndex = directionToIndex(dir);
    if (dirIndex < 0 || dirIndex >= 4) {
        dirIndex = directionToIndex(currentDirection);
    }
    if (dirIndex < 0 || dirIndex >= 4) {
        dirIndex = 1; // 默认向下
    }
    
    SpriteFrame* frame = standSprites[dirIndex];
    if (!frame) {
        CCLOG("Stand sprite is null for direction %d, using placeholder", dirIndex);
        frame = getSpriteFrame();
        if (!frame) {
            return;
        }
    }
    
    setSpriteFrame(frame);
    currentDirection = dir;
}
```

### 3. 修复 playWalkAnimation - 添加多层保护
**文件**: [Player.cpp](file:///c:/Users/emma/Documents/nyj_lxh_zlf_game/Classes/Core/Player.cpp)

```cpp
void Player::playWalkAnimation(Direction dir)
{
    if (isAnimationPlaying && currentDirection == dir)
    {
        return;
    }

    int dirIndex = directionToIndex(dir);
    if (dirIndex < 0 || dirIndex >= 4)
    {
        CCLOG("Invalid direction index: %d", dirIndex);
        showStandSprite(currentDirection);
        return;
    }

    if (walkAnimations[dirIndex].size() == 0)
    {
        CCLOG("No animation frames for direction %d, showing stand sprite", dirIndex);
        showStandSprite(dir);
        return;
    }

    if (currentAnimationAction)
    {
        stopAction(currentAnimationAction);
        currentAnimationAction = nullptr;
    }
    currentAnimation = nullptr;

    auto animation = Animation::createWithSpriteFrames(walkAnimations[dirIndex], 1.0f / animationFrameRate);
    if (!animation)
    {
        CCLOG("Failed to create animation, showing stand sprite");
        showStandSprite(dir);
        return;
    }
    
    currentAnimation = Animate::create(animation);
    if (!currentAnimation)
    {
        CCLOG("Failed to create animate, showing stand sprite");
        showStandSprite(dir);
        return;
    }
    
    currentAnimationAction = RepeatForever::create(currentAnimation);
    if (!currentAnimationAction)
    {
        CCLOG("Failed to create repeat action, showing stand sprite");
        currentAnimation = nullptr;
        showStandSprite(dir);
        return;
    }
    
    runAction(currentAnimationAction);
    isAnimationPlaying = true;
    currentDirection = dir;
}
```

void Player::stopAnimation()
{

        currentAnimationAction = nullptr;
    }
    currentAnimation = nullptr;
    isAnimationPlaying = false;
}
