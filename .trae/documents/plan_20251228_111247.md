## 修复方案：图片资源硬编码加载

### 问题诊断

图片资源存在于 `Resources/playerWalkImages/` 目录，但程序运行时工作目录可能是 `proj.win32/Debug.win32`，导致无法找到资源文件。

### 修复方案

#### 1. 修改 Player.cpp - 使用硬编码完整路径

**文件**: [Player.cpp](file:///c:/Users/emma/Documents/nyj_lxh_zlf_game/Classes/Core/Player.cpp)

将动态路径拼接改为硬编码完整路径：

```cpp
void Player::loadWalkAnimation(Direction dir, const std::string& path, int frameCount)
{
    int dirIndex = directionToIndex(dir);
    if (dirIndex < 0 || dirIndex >= 4) {
        CCLOG("Invalid direction in loadWalkAnimation: %d", dirIndex);
        return;
    }
    
    walkAnimations[dirIndex].clear();

    const char* Paths[4] = {
        "playerWalkImages/walkUp/",
        "playerWalkImages/walkDown/",
        "playerWalkImages/walkLeft/",
        "playerWalkImages/walkRight/"
    };
    
    for (int i = 0; i < frameCount; i++)
    {
        std::string frameName = std::string(Paths[dirIndex]) + std::to_string(i) + ".png";
        auto frame = SpriteFrame::create(frameName, Rect(0, 0, 70, 120));
        if (frame)
        {
            walkAnimations[dirIndex].pushBack(frame);
        }
        else
        {
            CCLOG("Failed to load animation frame: %s", frameName.c_str());
        }
    }

    CCLOG("Loaded %d frames for direction %d", (int)walkAnimations[dirIndex].size(), dirIndex);
}

void Player::loadStandSprite(Direction dir, const std::string& path)
{
    int dirIndex = directionToIndex(dir);
    if (dirIndex < 0 || dirIndex >= 4) {
        CCLOG("Invalid direction in loadStandSprite: %d", dirIndex);
        return;
    }
    
    const char* StandPaths[4] = {
        "playerWalkImages/standUp.png",
        "playerWalkImages/standDown.png",
        "playerWalkImages/standLeft.png",
        "playerWalkImages/standRight.png"
    };
    
    standSprites[dirIndex] = SpriteFrame::create(StandPaths[dirIndex], Rect(0, 0, 70, 120));
    if (!standSprites[dirIndex])
    {
        CCLOG("Failed to load stand sprite: %s", StandPaths[dirIndex]);
    }
}
```

#### 2. 修改 AppDelegate.cpp - 设置正确的资源搜索路径

**文件**: [AppDelegate.cpp](file:///c:/Users/emma/Documents/nyj_lxh_zlf_game/Classes/Core/AppDelegate.cpp)

在程序启动时设置资源搜索路径：

```cpp
#include "AppDelegate.h"
#include "MainMenuScene.h"

USING_NS_CC;

static cocos2d::Size designResolutionSize = cocos2d::Size(1136, 640);

AppDelegate::AppDelegate() {
}

AppDelegate::~AppDelegate() {
}

bool AppDelegate::applicationDidFinishLaunching() {
    auto director = Director::getInstance();
    auto glview = director->getOpenGLView();
    if(!glview) {
#if (CC_TARGET_PLATFORM == CC_PLATFORM_WIN32)
        glview = GLViewImpl::createWithRect("nyj_lxh_zlf_game", cocos2d::Rect(0, 0, designResolutionSize.width, designResolutionSize.height));
#else
        glview = GLViewImpl::create("nyj_lxh_zlf_game");
#endif
        director->setOpenGLView(glview);
    }

    director->setOpenGLView(glview);
    
    // 设置设计分辨率
    glview->setDesignResolutionSize(designResolutionSize.width, designResolutionSize.height, ResolutionPolicy::SHOW_ALL);
    
    // 添加资源搜索路径
    auto fileUtils = FileUtils::getInstance();
    std::vector<std::string> searchPaths;
    
    // 添加 Resources 目录到搜索路径
    searchPaths.push_back("Resources");
    
    // 添加项目根目录（确保能找到 Resources）
    std::string projectRoot = FileUtils::getInstance()->getDefaultResourceRootPath();
    if (!projectRoot.empty()) {
        searchPaths.push_back(projectRoot);
    }
    
    fileUtils->setSearchPaths(searchPaths);
    
    // 输出搜索路径以便调试
    auto paths = fileUtils->getSearchPaths();
    CCLOG("Resource search paths:");
    for (const auto& path : paths) {
        CCLOG("  - %s", path.c_str());
    }

    director->setDisplayStats(true);
    director->setAnimationInterval(1.0f / 60);

    auto scene = MainMenuScene::createScene();
    director->runWithScene(scene);

    return true;
}
```
