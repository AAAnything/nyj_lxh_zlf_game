### 4. 修复directionToIndex函数
**文件**: [Player.cpp](file:///c:/Users/emma/Documents/nyj_lxh_zlf_game/Classes/Core/Player.cpp)

正确处理Direction::NONE的情况：
```cpp
int Player::directionToIndex(Direction dir)
{
    if (dir == Direction::NONE) {
        return -1;  // 返回无效索引
    }
    
    switch (dir)
    {
        case Direction::UP: return 0;
        case Direction::DOWN: return 1;
        case Direction::LEFT: return 2;
        case Direction::RIGHT: return 3;
        default: return 1; // 默认向下
    }
}
```

同时在调用处添加检查：
```cpp
void Player::showStandSprite(Direction dir)
{
    stopAnimation();
    
    int dirIndex = directionToIndex(dir);
    if (dirIndex < 0 || dirIndex >= 4) {
        CCLOG("Invalid direction index in showStandSprite: %d", dirIndex);
        // 使用当前方向
        dirIndex = directionToIndex(currentDirection);
        if (dirIndex < 0 || dirIndex >= 4) {
            return;  // 无法显示
        }
    }
    
    if (standSprites[dirIndex])
    {
        setSpriteFrame(standSprites[dirIndex]);
        currentDirection = dir;
    }
    else
    {
        CCLOG("No stand sprite for direction %d", dirIndex);
    }
}
```
